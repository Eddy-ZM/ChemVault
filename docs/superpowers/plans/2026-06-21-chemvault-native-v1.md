# ChemVault Native v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the confirmed native SwiftUI ChemVault v1 application with online record discovery and device-local saved records.

**Architecture:** The target uses focused SwiftUI feature folders on top of a small `ChemVaultAPI` `URLSession` client. A `LibraryStore` owns remote discovery state; a `SavedStore` owns local stable record keys behind a protocol that can be replaced by cloud sync. `RootView` selects a `TabView` on compact widths and a `NavigationSplitView` on regular widths, reusing the same feature views.

**Tech Stack:** Swift 6, SwiftUI, Foundation `URLSession`, Swift Testing, XCTest UI tests, iOS/iPadOS/macOS/visionOS target generated by Xcode 27.

---

```swift
enum ChemVaultAPIError: LocalizedError, Equatable {
    case invalidResponse
    case httpStatus(Int)
    case decoding
    case transport(String)
    var errorDescription: String? {
        switch self {
        case .invalidResponse: "The ChemVault service returned an invalid response."
        case let .httpStatus(status): "ChemVault is currently unavailable (HTTP \(status))."
        case .decoding: "ChemVault returned data the app could not read."
        case .transport: "Could not reach ChemVault. Check your connection and retry."
        }
    }
}
```

```swift
actor RecordingTransport: HTTPTransporting {
    enum Response { case success(Data); case failure(ChemVaultAPIError) }
    private(set) var lastRequest: URLRequest?
    let response: Response
    init(response: Response) { self.response = response }
    func send(_ request: URLRequest) async throws -> (Data, HTTPURLResponse) {
        lastRequest = request
        switch response {
        case let .success(data): return (data, HTTPURLResponse(url: request.url!, statusCode: 200, httpVersion: nil, headerFields: nil)!)
        case let .failure(error): throw error
        }
    }
}
```

```swift
protocol SavedRecordStore: AnyObject {
    var allKeys: [String] { get }
    func contains(_ key: String) -> Bool
    func toggle(_ key: String)
}

@MainActor
final class SavedStore: ObservableObject, SavedRecordStore {
    @Published private(set) var keys: Set<String>
    private let defaults: UserDefaults
    private let storageKey = "chemvault.saved-record-keys"

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        self.keys = Set(defaults.stringArray(forKey: storageKey) ?? [])
    }

    var allKeys: [String] { keys.sorted() }
    func contains(_ key: String) -> Bool { keys.contains(key) }
    func toggle(_ key: String) { keys.contains(key) ? keys.remove(key) : keys.insert(key); defaults.set(allKeys, forKey: storageKey) }
}
```
